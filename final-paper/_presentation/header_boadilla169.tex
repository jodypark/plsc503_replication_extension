% --- Custom footline with author | course | date + page x/y ---
\setbeamertemplate{footline}{%
  \leavevmode%
  \hbox{%
    % Left: Author name
    \begin{beamercolorbox}[wd=.333333\paperwidth,ht=3.05ex,dp=1ex,center]{author in head/foot}%
      \usebeamerfont{author in head/foot}\insertauthor
    \end{beamercolorbox}%
    % Middle: Course/institute
    \begin{beamercolorbox}[wd=.333333\paperwidth,ht=3.05ex,dp=1ex,center]{title in head/foot}%
      \usebeamerfont{title in head/foot}\insertshortinstitute
    \end{beamercolorbox}%
    % Right: Date and slide number x/y
    \begin{beamercolorbox}[wd=.333333\paperwidth,ht=3.05ex,dp=1ex,right]{date in head/foot}%
      \usebeamerfont{date in head/foot}%
      May 4, 2026\hspace*{2em}%
      \insertframenumber{}/\inserttotalframenumber\hspace*{2ex}%
    \end{beamercolorbox}%
  }%
  \vskip0pt%
}
